  1#!/usr/bin/env python
  2"""
  3nav_pkg.action_client_node
  4==========================
  5
  6A ROS action client node that interfaces with the `/reaching_goal` action server.
  7
  8This node allows setting and cancelling navigation goals, publishing robot
  9state (position and velocity), and tracking the last target set.
 10
 11Author:
 12    Luca Carpaneto
 13
 14Version:
 15    1.0
 16
 17Date:
 18    13/10/2025
 19"""
 20
 21import rospy
 22import actionlib
 23from assignment_2_2024.msg import PlanningAction, PlanningGoal
 24from nav_msgs.msg import Odometry
 25from geometry_msgs.msg import Point, PoseStamped
 26from nav_pkg.msg import robot_pos_vel
 27from nav_pkg.srv import SendTarget, SendTargetResponse
 28from std_srvs.srv import SetBool, SetBoolResponse
 29import threading
 30
 31#: The most recent target position sent to the action server.
 32_last_target = None
 33
 34#: Threading lock to protect access to the global `_last_target`.
 35_last_target_lock = threading.Lock()
 36
 37
 38class ActionClientNode:
 39    """
 40    A ROS node that acts as an action client to send navigation goals.
 41
 42    This class handles:
 43        - Connection to the `/reaching_goal` action server
 44        - `/set_target` and `/cancel_target` services
 45        - Publishing robot position, velocity, and last sent target
 46    """
 47
 48    def __init__(self):
 49        """
 50        Initialize the action client node, publishers, subscribers, and services.
 51        """
 52        rospy.init_node('action_client_node')
 53
 54        #: Action client for communicating with the `/reaching_goal` action server.
 55        self.client = actionlib.SimpleActionClient('/reaching_goal', PlanningAction)
 56        rospy.loginfo("Waiting for action server '/reaching_goal'...")
 57        self.client.wait_for_server()
 58        rospy.loginfo("Connected to '/reaching_goal' action server.")
 59
 60        #: Publisher for robot position and velocity.
 61        self.robot_state_pub = rospy.Publisher('/robot_state', robot_pos_vel, queue_size=10)
 62
 63        #: Publisher for the last target position.
 64        self.last_target_pub = rospy.Publisher('/last_target', Point, queue_size=1, latch=True)
 65
 66        #: Subscriber for odometry data.
 67        rospy.Subscriber('/odom', Odometry, self.odom_callback)
 68
 69        #: Service to set a new navigation target.
 70        self.set_target_srv = rospy.Service('/set_target', SendTarget, self.set_target_callback)
 71
 72        #: Service to cancel the current navigation goal.
 73        self.cancel_srv = rospy.Service('/cancel_target', SetBool, self.cancel_callback)
 74
 75        #: Current odometry position.
 76        self.current_odom = None
 77
 78        #: Current velocity (twist).
 79        self.current_twist = None
 80
 81        rospy.loginfo("action_client_node is ready. Use /set_target to send goals.")
 82
 83    def odom_callback(self, msg):
 84        """
 85        Handle incoming odometry messages and publish the robot's position and velocity.
 86
 87        Parameters
 88        ----------
 89        msg : nav_msgs.msg.Odometry
 90            The received odometry message.
 91        """
 92        self.current_odom = msg.pose.pose.position
 93        self.current_twist = msg.twist.twist
 94
 95        rs = robot_pos_vel()
 96        rs.x = self.current_odom.x
 97        rs.y = self.current_odom.y
 98        rs.vel_x = self.current_twist.linear.x
 99        rs.vel_y = self.current_twist.linear.y
100        self.robot_state_pub.publish(rs)
101
102    def set_target_callback(self, req):
103        """
104        Handle `/set_target` service calls to send a new navigation goal.
105
106        Parameters
107        ----------
108        req : nav_pkg.srv.SendTargetRequest
109            The service request containing target coordinates.
110
111        Returns
112        -------
113        nav_pkg.srv.SendTargetResponse
114            True if the goal was accepted.
115        """
116        global _last_target, _last_target_lock
117
118        rospy.loginfo(f"New target received: x={req.x:.2f}, y={req.y:.2f}")
119
120        # Create action goal
121        goal = PlanningGoal()
122        goal.target_pose = PoseStamped()
123        goal.target_pose.pose.position.x = req.x
124        goal.target_pose.pose.position.y = req.y
125        goal.target_pose.pose.position.z = 0.0
126
127        # Save and publish last target
128        with _last_target_lock:
129            _last_target = Point(req.x, req.y, 0.0)
130            self.last_target_pub.publish(_last_target)
131
132        # Send goal
133        self.client.send_goal(goal,
134                              done_cb=self.done_cb,
135                              active_cb=self.active_cb,
136                              feedback_cb=self.feedback_cb)
137        return SendTargetResponse(accepted=True)
138
139    def cancel_callback(self, req):
140        """
141        Handle `/cancel_target` service calls to cancel the current navigation goal.
142
143        Parameters
144        ----------
145        req : std_srvs.srv.SetBoolRequest
146            The service request containing a boolean flag.
147
148        Returns
149        -------
150        std_srvs.srv.SetBoolResponse
151            Result of the cancel operation.
152        """
153        if req.data:
154            rospy.loginfo("Cancelling current goal...")
155            self.client.cancel_all_goals()
156            return SetBoolResponse(success=True, message="Goal cancelled.")
157        else:
158            return SetBoolResponse(success=True, message="No cancellation requested.")
159
160    def active_cb(self):
161        """Callback executed when the action goal becomes active."""
162        rospy.loginfo("Goal is now active.")
163
164    def feedback_cb(self, feedback):
165        """
166        Callback executed when feedback is received from the action server.
167
168        Parameters
169        ----------
170        feedback : assignment_2_2024.msg.PlanningFeedback
171            The feedback message from the action server.
172        """
173        rospy.loginfo_throttle(1.0, f"Feedback: {feedback.stat}")
174
175    def done_cb(self, state, result):
176        """
177        Callback executed when the action goal completes.
178
179        Parameters
180        ----------
181        state : int
182            The final state of the goal.
183        result : assignment_2_2024.msg.PlanningResult
184            The result message from the action server.
185        """
186        rospy.loginfo(f"Goal finished with state [{state}]")
187
188    def spin(self):
189        """Keep the node alive and responsive."""
190        rospy.spin()
191
192
193if __name__ == '__main__':
194    """
195    Main entry point for the ROS action client node.
196    """
197    node = ActionClientNode()
198    node.spin()
