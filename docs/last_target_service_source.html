 1#!/usr/bin/env python
 2"""
 3ROS Node: last_target_service_node
 4==================================
 5
 6This node provides a service `/get_last_target` that returns the last target
 7position received from the `/last_target` topic.
 8
 9The service is useful for other nodes that need to know the most recent target
10set by the user or another node (e.g., an action client).
11
12**Published Topics**
13    None
14
15**Subscribed Topics**
16    - `/last_target` (`geometry_msgs/Point`): Receives the last target position.
17
18**Services**
19    - `/get_last_target` (`nav_pkg/GetLastTarget`): Returns the last known target position.
20
21"""
22
23import rospy
24from nav_pkg.srv import GetLastTarget, GetLastTargetResponse
25from geometry_msgs.msg import Point
26
27_last_target = None
28
29
30def last_target_callback(msg):
31    """
32    Callback for the `/last_target` topic.
33
34    This function updates the global variable `_last_target` whenever a new
35    `geometry_msgs/Point` message is received.
36
37    Parameters
38    ----------
39    msg : geometry_msgs.msg.Point
40        The latest target position published by the `action_client_node`.
41    """
42    global _last_target
43    _last_target = msg
44
45
46def handle_get_last_target(req):
47    """
48    Handle service requests to `/get_last_target`.
49
50    Returns the most recent target position received via the `/last_target`
51    topic. If no target has been received yet, it returns `(0.0, 0.0)` and
52    logs a warning.
53
54    Parameters
55    ----------
56    req : nav_pkg.srv.GetLastTargetRequest
57        Empty service request (no parameters).
58
59    Returns
60    -------
61    nav_pkg.srv.GetLastTargetResponse
62        Response containing the `x` and `y` coordinates of the last target.
63    """
64    global _last_target
65    if _last_target is None:
66        rospy.logwarn("No target has been set yet; returning (0,0)")
67        return GetLastTargetResponse(x=0.0, y=0.0)
68    else:
69        return GetLastTargetResponse(x=_last_target.x, y=_last_target.y)
70
71
72def main():
73    """
74    Initialize and run the `last_target_service_node`.
75
76    This function sets up:
77      - A ROS subscriber to `/last_target` to track the last received target.
78      - A ROS service `/get_last_target` to provide the stored target on demand.
79
80    The node runs continuously using `rospy.spin()` until shut down.
81    """
82    rospy.init_node('last_target_service_node')
83    rospy.Subscriber('/last_target', Point, last_target_callback)
84    rospy.Service('/get_last_target', GetLastTarget, handle_get_last_target)
85    rospy.loginfo("Service /get_last_target is ready.")
86    rospy.spin()
87
88
89if __name__ == '__main__':
90    main()
